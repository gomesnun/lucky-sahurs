rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // ---- Saves: cada conta só vê e mexe nos SEUS slots ----
    match /users/{uid}/saves/{slot} {
      function validSave() {
        let d = request.resource.data;
        return d.keys().hasOnly(['coins', 'total_earned', 'playtime', 'rolls', 'rebirths', 'data'])
            // "rebirths" é opcional: versões antigas do jogo não o enviam e continuam a poder gravar
            && d.keys().hasAll(['coins', 'total_earned', 'playtime', 'rolls', 'data'])
            && d.coins is number && d.total_earned is number
            && d.playtime is number && d.rolls is number
            && d.coins >= 0 && d.total_earned >= 0 && d.playtime >= 0 && d.rolls >= 0
            && (!('rebirths' in d) || (d.rebirths is int && d.rebirths >= 0 && d.rebirths <= 1000))
            && d.data is string && d.data.size() < 900000;
      }
      allow read, delete: if request.auth != null && request.auth.uid == uid;
      allow create, update: if request.auth != null && request.auth.uid == uid
                            && slot in ['slot1', 'slot2', 'slot3']
                            && validSave();
    }
    // ---- Sessão: uma conta, um jogo de cada vez ----
    // Marca de "estou a jogar", renovada de 30 em 30 s pelo jogo. Só o dono a vê e a escreve, e só se
    // pode tirar a marca a outro PC se ele já a largou (active = false) ou se deixou de a renovar há mais
    // de 90 s (SESSION_STALE no online/firebase.py — os dois valores têm de ser iguais).
    match /users/{uid}/session/current {
      function validLock() {
        let d = request.resource.data;
        return d.keys().hasOnly(['session_id', 'active', 'heartbeat'])
            && d.keys().hasAll(['session_id', 'active', 'heartbeat'])
            // nome da instalação: o jogo usa 32 caracteres hexadecimais (uuid4)
            && d.session_id is string && d.session_id.matches('[A-Za-z0-9_-]{8,64}')
            && d.active is bool
            // a hora tem de ser a do servidor: assim não dá para fingir uma marca fresca (para trancar
            // a conta a alguém) nem uma marca velha (para roubar a marca a quem está mesmo a jogar)
            && d.heartbeat == request.time;
      }
      allow read:   if request.auth != null && request.auth.uid == uid;
      allow create: if request.auth != null && request.auth.uid == uid && validLock();
      allow update: if request.auth != null && request.auth.uid == uid && validLock()
                    && (resource.data.session_id == request.resource.data.session_id
                        || resource.data.active == false
                        || request.time > resource.data.heartbeat + duration.value(90, 's'));
      // apagar não: um "delete" não traz session_id nenhum, por isso não havia forma de distinguir
      // quem larga a sua marca de quem estava a limpar a marca de outro para entrar por cima
      allow delete: if false;
    }
    // ---- Leaderboard: toda a gente lê, cada conta escreve só a sua linha ----
    match /leaderboard/{uid} {
      function validEntry() {
        let d = request.resource.data;
        return d.keys().hasOnly(['username', 'coins', 'playtime', 'rolls', 'rebirths', 'updated_at'])
            // "rolls" e "rebirths" são opcionais: uma versão antiga do jogo continua a publicar dinheiro/tempo
            && d.keys().hasAll(['username', 'coins', 'playtime', 'updated_at'])
            // o nome tem de ser o da própria conta (o email interno é <username>@luckysahurs.invalid)
            && d.username is string
            // mesmo formato que o jogo aceita: 3 a 16 letras minúsculas, números ou _
            && d.username.matches('[a-z0-9_]{3,16}')
            && request.auth.token.email == d.username + '@luckysahurs.invalid'
            && d.coins is number && d.coins >= 0
            && d.playtime is number && d.playtime >= 0 && d.playtime < 100000000
            // o dinheiro tem de caber no tempo jogado (folga de 1e15/s: com rebirths o rendimento passa o
            // máximo antigo de ~5,9e13/s, e os ganhos offline dão dinheiro sem somar tempo jogado)
            && d.coins <= d.playtime * 1000000000000000.0 + 100000000000000.0
            // rolls: o jogo faz no máximo 50 rolls por frame (=3000/s); aqui 4000/s + folga
            && (!('rolls' in d) || (d.rolls is int && d.rolls >= 0
                                    && d.rolls <= d.playtime * 4000.0 + 5000.0))
            // rebirths: o n-ésimo rebirth custa 500000 * 4^(n-1) em moedas na mão, e o total ganho nunca
            // desce, por isso quem tem n rebirths já ganhou pelo menos 500000 * (4^n - 1) / 3 (10% de folga)
            && (!('rebirths' in d) || (d.rebirths is int && d.rebirths >= 0 && d.rebirths <= 1000
                                       && d.coins >= 0.9 * 500000.0 * (math.pow(4, d.rebirths) - 1) / 3.0))
            // a hora tem de ser a do servidor (não dá para falsificar)
            && d.updated_at == request.time;
      }
      allow read: if true;
      allow create: if request.auth != null && request.auth.uid == uid && validEntry();
      // no máximo uma atualização de 5 em 5 minutos por conta (o jogo publica de 10 em 10 minutos,
      // com um mínimo de 6 entre publicações: a regra tem de pedir MENOS do que isso)
      allow update: if request.auth != null && request.auth.uid == uid && validEntry()
                    && request.time > resource.data.updated_at + duration.value(5, 'm');
      allow delete: if false;
    }
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
